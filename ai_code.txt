function addAIMsg(text, isUser){
  const msgs = document.getElementById('aiMessages');
  const div = document.createElement('div');
  div.className = 'ai-msg ' + (isUser ? 'user' : 'assistant');
  div.innerHTML = '<div class="bubble"><div class="ai-bubble">' + text + '</div></div>';
  msgs.appendChild(div);
  msgs.scrollTop = msgs.scrollHeight;
}

function addTyping(){

function removeTyping(){
  const t = document.getElementById('aiTyping');
  if(t) t.remove();
}

function toggleVoice(){
  if(!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)){
    toast('浏览器不支持语音输入');
    return;
  }
  if(isRecording){
    recognition.stop();
    return;
  }
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  recognition = new SR();
  recognition.lang = 'zh-CN';
  recognition.continuous = false;
  recognition.interimResults = false;
  recognition.onstart = function(){
    isRecording = true;
    document.getElementById('aiVoiceBtn').classList.add('recording');
  };
  recognition.onresult = function(e){
    const text = e.results[0][0].transcript;
    document.getElementById('aiInput').value = text;
    sendAI();
  };
  recognition.onerror = function(e){
    toast('语音识别失败');
  };
  recognition.onend = function(){
    isRecording = false;
    document.getElementById('aiVoiceBtn').classList.remove('recording');
  };
  recognition.start();
}

function buildSystemPrompt(){
  const students = state.students.map(s => s.name + '(第' + s.group + '组)').join('、');
  const today = curDateStr();
  return '你是班级管理助手，帮老师快速录入分数和查数据。' +
    '当前日期：' + today + '。' +
    '学生名单：' + students + '。' +
    '你需要把用户的指令转换成JSON格式返回，不要返回其他内容。' +
    '支持的操作：' +
    '1. 加扣分：{"action":"add","student":"姓名","value":2,"label":"表现好"}' +
    '2. 查数据：{"action":"query","question":"问题"}' +
    '3. 删除记录：{"action":"delete","student":"姓名"}' +
    '只返回JSON，不要解释。';
}

async function sendAI(){
  const input = document.getElementById('aiInput');
  const text = input.value.trim();
  if(!text) return;
  input.value = '';
  addAIMsg(text, true);
  addTyping();

  try{
    const resp = await fetch(DASHSCOPE_URL, {
      method: 'POST',
      headers: {
        'Authorization': 'Bearer ' + DASHSCOPE_API_KEY,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        model: 'qwen-turbo',
        messages: [
          {role: 'system', content: buildSystemPrompt()},
          {role: 'user', content: text}
        ],
        temperature: 0.3
      })
    });
    const data = await resp.json();
    const aiText = data.choices[0].message.content.trim();
    removeTyping();

    try{
      const action = JSON.parse(aiText);
      executeAIAction(action, text);
    }catch(e){
      addAIMsg(aiText, false);
    }
  }catch(e){
    removeTyping();
    addAIMsg('出错了：' + e.message, false);
  }
}

async function executeAIAction(action, originalText){
  if(action.action === 'add'){
    const student = state.students.find(s => s.name === action.student);
    if(!student){
      addAIMsg('没找到学生：' + action.student, false);
      return;
    }
    await addScore(student.id, action.value, action.label || 'AI录入');
    addAIMsg('已完成：' + action.student + ' ' + (action.value > 0 ? '+' : '') + action.value + '分', false);
  }else if(action.action === 'query'){
    addAIMsg('查询：' + action.question, false);
    const result = queryLocalData(action.question);
    addAIMsg(result, false);
  }else if(action.action === 'delete'){
    const student = state.students.find(s => s.name === action.student);
    if(!student){
      addAIMsg('没找到学生：' + action.student, false);
      return;
    }
    if(disDay[student.id]){
      disDay[student.id] = [];
      await saveDisToGroup(student.id, []);
      await loadDayData(curDateStr());
      renderDis();
      addAIMsg('已删除：' + action.student + ' 今天的所有记录', false);
    }else{
      addAIMsg(action.student + ' 今天没有记录', false);
    }
  }else{
    addAIMsg(originalText, false);
  }
}

function queryLocalData(question){
  const q = question.toLowerCase();
  if(q.includes('扣分') || q.includes('减分') || q.includes('表现不好')){
    let result = '今天扣分情况：\n';
    let has = false;
    for(const sid in disDay){
      const recs = disDay[sid];
      const neg = recs.filter(r => r.value < 0);
      if(neg.length > 0){
        const student = state.students.find(s => s.id === sid);
        if(student){
          result += student.name + '：' + neg.map(r => r.label + ' ' + r.value + '分').join('、') + '\n';
          has = true;
        }
      }
    }
    return has ? result : '今天没有人扣分';
  }
  if(q.includes('加分') || q.includes('表扬') || q.includes('表现好')){
    let result = '今天加分情况：\n';
    let has = false;
    for(const sid in disDay){
      const recs = disDay[sid];
      const pos = recs.filter(r => r.value > 0);
      if(pos.length > 0){
        const student = state.students.find(s => s.id === sid);
        if(student){
          result += student.name + '：' + pos.map(r => r.label + ' +' + r.value + '分').join('、') + '\n';
          has = true;
        }
      }
    }
    return has ? result : '今天没有人加分';
  }
  if(q.includes('总结') || q.includes('汇总') || q.includes('情况')){
    let result = '今日纪律汇总：\n';
    let total = 0;
    for(const sid in disDay){
      const recs = disDay[sid];
      const sum = recs.reduce((a, r) => a + r.value, 0);
      if(sum !== 0){
        const student = state.students.find(s => s.id === sid);
        if(student){
          result += student.name + '：' + (sum > 0 ? '+' : '') + sum + '分\n';
          total += sum;
        }
      }
    }
    result += '全班合计：' + (total > 0 ? '+' : '') + total + '分';
    return result;
  }
  return '我理解的问题是：' + question + '\n你可以试试问"今天谁扣分了"或"今天谁加分了"';
}
